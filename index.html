// Variable global para mantener el historial y la memoria de la conversación let chatHistory = []; async function procesarComandoIA(textoForzado = null) { const input = document.getElementById('global-command-input'); const texto = textoForzado || (input ? input.value.trim() : ''); if(!texto) return; switchTab('chat'); const box = document.getElementById('chat-history-box'); // Mostrar mensaje del usuario en pantalla if(box) { box.innerHTML += `
${typeof usuarioActual !== 'undefined' ? usuarioActual : 'Usuario'}: ${texto}
`; input.value = ''; box.scrollTop = box.scrollHeight; } let respuestaIA = ""; try { // Petición real al backend seguro en Vercel const response = await fetch('https://orion-la-sooty.vercel.app/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: texto, history: chatHistory }) }); const data = await response.json(); if (response.ok && data.reply) { respuestaIA = data.reply; // Guardar en memoria para mantener contexto de la charla chatHistory.push({ role: 'user', text: texto }); chatHistory.push({ role: 'model', text: respuestaIA }); } else { respuestaIA = "Disculpame Héctor, tuve un problemita técnico. Volvé a intentarlo."; } } catch (error) { console.error("Error de red:", error); respuestaIA = "Che Héctor, no me pude conectar con el servidor en este momento."; } // Mostrar respuesta real de la IA en el chat if(box) { box.innerHTML += `
ORION: ${respuestaIA}
`; box.scrollTop = box.scrollHeight; } // Enviar exactamente esta respuesta al sistema de voz intacto if (typeof hablarOrion === 'function') { hablarOrion(respuestaIA); } }
